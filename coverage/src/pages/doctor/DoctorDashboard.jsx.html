import { useEffect, useState } from 'react'
import { Link } from 'react-router-dom'
import { getDoctorAppointments } from '../../api/doctor'
import AppointmentStatus from '../../components/AppointmentStatus'
const asList = (data) => Array.isArray(data) ? data : data?.appointments || data?.results || []
export default function DoctorDashboard() { const [items, setItems] = useState([]); const [error, setError] = useState(''); useEffect(() => { getDoctorAppointments().then((data) => setItems(asList(data).sort((a,b) => `${a.date}${a.scheduled_time}`.localeCompare(`${b.date}${b.scheduled_time}`)))).catch(() => setError('Could not load upcoming appointments.')) }, []); return <div className="mx-auto max-w-5xl px-5 py-10 sm:px-6 sm:py-14"><p className="text-xs font-semibold uppercase tracking-[.16em] text-ink/55">Doctor portal</p><h1 className="mt-2 font-display text-4xl font-bold">Upcoming appointments</h1><p className="mt-3 text-sm text-ink/65">Open an appointment to record diagnosis and treatment after the consultation.</p>{error && <p className="mt-5 border-l-4 border-status-alert bg-status-alert/10 p-3 text-sm">{error}</p>}<div className="mt-8 space-y-3">{items.length ? items.map((item) => <article key={item.id} className="flex flex-wrap items-center justify-between gap-4 border border-ink/15 bg-panel p-5"><div><p className="font-semibold">{item.patient_username}</p><p className="mt-1 text-sm text-ink/60">{item.date} · {item.scheduled_time?.slice(0, 5)}</p></div><div className="flex items-center gap-4"><AppointmentStatus status={item.status}/><Link className="text-sm font-semibold underline" to={`/doctor/appointments/${item.id}`} state={{ appointment: item }}>Open record</Link></div></article>) : <p className="border border-ink/15 bg-panel p-5 text-sm text-ink/65">No upcoming appointments.</p>}</div></div> }
