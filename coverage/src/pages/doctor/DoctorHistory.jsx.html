import { useEffect, useState } from 'react'
import { getDoctorHistory } from '../../api/doctor'
const asList = (data) => Array.isArray(data) ? data : data?.appointments || data?.results || []
export default function DoctorHistory() { const [items, setItems] = useState([]); const [error, setError] = useState(''); useEffect(() => { getDoctorHistory().then((data) => setItems(asList(data))).catch(() => setError('Could not load completed appointments.')) }, []); return <div className="mx-auto max-w-5xl px-5 py-10 sm:px-6 sm:py-14"><p className="text-xs font-semibold uppercase tracking-[.16em] text-ink/55">Doctor portal</p><h1 className="mt-2 font-display text-4xl font-bold">Completed appointment history</h1>{error && <p className="mt-5 border-l-4 border-status-alert bg-status-alert/10 p-3 text-sm">{error}</p>}<div className="mt-8 space-y-3">{items.length ? items.map((item) => <article key={item.id} className="border border-ink/15 bg-panel p-5"><p className="font-semibold">{item.patient_username}</p><p className="mt-1 text-sm text-ink/60">{item.date} · {item.scheduled_time?.slice(0, 5)}</p><p className="mt-4 text-sm"><b>Diagnosis:</b> {item.diagnosis || 'Not recorded'}</p><p className="mt-2 text-sm"><b>Treatment:</b> {item.treatment || 'Not recorded'}</p></article>) : <p className="border border-ink/15 bg-panel p-5 text-sm text-ink/65">No completed appointments.</p>}</div></div> }
