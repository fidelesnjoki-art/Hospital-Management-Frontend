import { useState } from 'react'
import { Link, useNavigate } from 'react-router-dom'
import { login } from '../../api/auth'
import { useAuth } from '../../context/AuthContext'
import SiteShell from '../../components/SiteShell'
const destination = (role) => role === 'admin' ? '/admin' : role === 'doctor' ? '/doctor' : '/patient'
export default function Login() { const [username, setUsername] = useState(''), [password, setPassword] = useState(''), [error, setError] = useState(''), [loading, setLoading] = useState(false); const { refreshUser } = useAuth(); const navigate = useNavigate(); async function submit(e) { e.preventDefault(); setLoading(true); setError(''); try { await login(username, password); const user = await refreshUser(); if (!user) throw new Error(); navigate(destination(user.role)) } catch (err) { setError(err.response?.data?.detail || 'Unable to sign in with those details.') } finally { setLoading(false) } } return <SiteShell><div className="flex min-h-[calc(100vh-180px)] items-center justify-center bg-paper p-5"><form onSubmit={submit} className="site-card w-full max-w-sm p-7 sm:p-9"><p className="site-eyebrow">Welcome back</p><h1 className="mt-2 font-display text-3xl font-bold">Sign in</h1><p className="mt-2 text-sm text-ink/60">Use your email or username to access your account.</p>{error && <p className="site-alert mt-5">{error}</p>}<label className="mt-6 block text-sm font-semibold">Email or username<input required value={username} onChange={(e) => setUsername(e.target.value)} className="mt-2 w-full px-3 py-3 font-normal" /></label><label className="mt-5 block text-sm font-semibold">Password<input required type="password" value={password} onChange={(e) => setPassword(e.target.value)} className="mt-2 w-full px-3 py-3 font-normal" /></label><button disabled={loading} className="site-button site-button-primary mt-7 w-full">{loading ? 'Signing in...' : 'Sign in'}</button><p className="mt-6 text-center text-sm text-ink/65">Need an account? <Link to="/signup" className="font-semibold underline">Sign up</Link></p></form></div></SiteShell> }
